composed overview: whole-classification isiccomb -> isic.3 structure

Motivation

Component (d): the composed, whole-classification isiccomb -> isic.3 view – one graph, not per country-year, showing every place in the classification where a real one-to-many split or many-to-one aggregation happens, with the pass-through-unchanged majority collapsed out of view.

Composed with xmap::compose_xmap() in Extraction, chaining isiccomb -> isic and isic -> isic.3 through their shared isic intermediate into a single isiccomb -> isic.3 crossmap – without materialising isic-level values, which aren’t meaningful on their own (they’re an artifact of the naive split). This is the same information Explorer/ Country variants show per country-year component, at the whole-classification level: not “what happened in one country-year” but “which isiccomb codes, anywhere in the dataset, actually carry a real split or aggregation.”

Setup

Reads the crossmap collection prepared by Extraction – render that page first if data/interim/crossmap_collection.rds is missing or stale.

Prep data for the diagram

Every isiccomb code turns out to compose the exact same way in every country-year that reports it (see Extraction), so the composed crossmap collapses to one global structure with no extra reconciliation – distinct (isiccomb, isic3, weight) triples, bundling away the pass-through-unchanged majority (a code mapping to exactly one isic.3 parent at weight 1) and showing only codes that cross an isic.3 boundary.

Code 
## dedup: every isiccomb code composes identically in every country-year
## that reports it, so a global distinct() is a lossless reduction
composed_code_edges <- composed_isiccomb_isic3 |>
  distinct(isiccomb, isic3, weight) |>
  add_count(isiccomb, name = "n_isic3")

n_codes <- n_distinct(composed_code_edges$isiccomb)
n_bundled <- composed_code_edges |> filter(n_isic3 == 1) |> nrow()
n_crossing <- n_codes - n_bundled

composed_crossing_edges <- composed_code_edges |>
  filter(n_isic3 > 1) |>
  arrange(desc(n_isic3), isiccomb)

Node-link overview

Code 
d3 = require("d3@7")
Code 
edges = transpose(composed_crossing_edges)

codeGroups = d3.sort(
  d3.groups(edges, d => d.isiccomb),
  ([, rows]) => -rows.length
)

function evenY(n, height, pad = 18) {
  if (n <= 1) return [height / 2];
  const usable = height - 2 * pad;
  return d3.range(n).map(i => pad + (i * usable) / (n - 1));
}

// fixed [0, 1] domain -- weights here are already fractions of the source
// isiccomb code's value, so "how dark" (how much of this code's value
// lands here) means the same thing in every small multiple.
shade = d3.scaleSequential(d3.interpolateBlues).domain([0, 1])

function buildComposedNodeLink(edgesForCode) {
  const isiccomb = edgesForCode[0].isiccomb;

  const col0 = [isiccomb];
  const col1 = d3.sort(edgesForCode, d => -d.weight).map(d => d.isic3);

  const width = 220;
  const height = Math.max(60, col1.length * 20 + 24);

  const x0 = 54, x1 = width - 54;
  const y0 = evenY(1, height);
  const y1 = evenY(col1.length, height);

  const pos0 = new Map(col0.map((d, i) => [d, y0[i]]));
  const pos1 = new Map(col1.map((d, i) => [d, y1[i]]));

  const svg = d3.create("svg")
    .attr("viewBox", [0, 0, width, height])
    .attr("width", width)
    .attr("height", height)
    .attr("style", "max-width: 100%; height: auto; font: 10px sans-serif;");

  // isiccomb -> isic.3, dashed (splitting), weight labelled
  svg.append("g")
    .selectAll("path")
    .data(edgesForCode)
    .join("path")
      .attr("d", d => {
        const sy = pos0.get(d.isiccomb), ty = pos1.get(d.isic3);
        return `M${x0},${sy} C${(x0 + x1) / 2},${sy} ${(x0 + x1) / 2},${ty} ${x1},${ty}`;
      })
      .attr("fill", "none")
      .attr("stroke", "#888")
      .attr("stroke-width", 1)
      .attr("stroke-dasharray", "3,2")
    .append("title")
      .text(d => `${d.isiccomb} -> ${d.isic3}\nweight: ${d.weight.toFixed(3)}`);

  svg.append("g")
    .selectAll("text")
    .data(edgesForCode)
    .join("text")
      .attr("x", (x0 + x1) / 2)
      .attr("y", d => (pos0.get(d.isiccomb) + pos1.get(d.isic3)) / 2 - 2)
      .attr("text-anchor", "middle")
      .attr("fill", "#666")
      .text(d => d.weight.toFixed(2));

  svg.append("circle").attr("cx", x0).attr("cy", y0[0]).attr("r", 3).attr("fill", "#333");
  svg.append("text")
    .attr("x", x0 - 6).attr("y", y0[0]).attr("dy", "0.32em")
    .attr("text-anchor", "end").attr("font-style", "italic").attr("font-weight", 600)
    .text(isiccomb);

  svg.append("g")
    .selectAll("circle")
    .data(col1)
    .join("circle")
      .attr("cx", x1).attr("cy", d => pos1.get(d)).attr("r", 6)
      .attr("fill", d => shade(edgesForCode.find(e => e.isic3 === d).weight))
      .attr("stroke", "#333").attr("stroke-width", 0.5)
    .append("title")
      .text(d => `${d} -- receives ${edgesForCode.find(e => e.isic3 === d).weight.toFixed(3)} of ${isiccomb}'s value`);
  svg.append("g")
    .selectAll("text")
    .data(col1)
    .join("text")
      .attr("x", x1 + 10).attr("y", d => pos1.get(d)).attr("dy", "0.32em")
      .attr("text-anchor", "start")
      .text(d => d);

  return svg.node();
}

nodeLinkOverview = html`<div style="display:flex; flex-wrap:wrap; gap:1rem; align-items:flex-start;">
  ${codeGroups.map(([isiccombCode, rows]) => {
    return html`<div style="border:1px solid #ddd; border-radius:6px; padding:0.5rem;">
      <div style="font: 600 12px sans-serif; margin-bottom:4px;">${isiccombCode} (${rows.length} isic.3 parents)</div>
      ${buildComposedNodeLink(rows)}
    </div>`;
  })}
</div>`
